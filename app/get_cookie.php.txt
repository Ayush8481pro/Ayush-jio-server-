<?php
error_reporting(0);
include "functions.php";

header("Content-Type: text/plain");
header("Access-Control-Allow-Origin: *");

$id = htmlspecialchars($_REQUEST['id'] ?? '');
$cooks = htmlspecialchars($_REQUEST['ck'] ?? '');

if (empty($cooks) || empty($id)) {
    http_response_code(400);
    exit("Missing required parameters");
}

$cookie = hex2bin($cooks);
$headers = [
    'Cookie: ' . $cookie,
    'Content-Type: application/x-www-form-urlencoded',
    'User-Agent: plaYtv/7.1.3 (Linux;Android 14) ExoPlayerLib/2.11.7'
];

$chs = explode('-', $id);
$url = sprintf("https://jiotvmblive.cdn.jio.com/bpk-tv/%s/Fallback/%s", $chs[0], $id);

// Step 1: Make an initial request to the playlist URL.
// This mimics the original script and may be necessary for the CDN to set the __hdnea__ cookie.
cUrlGetData($url, $headers);

// Step 2: Now fetch the fresh cookie (this makes a second HTTP request).
try {
    $freshCookieHex = get_and_refresh_cookie($url, $headers);
    echo $freshCookieHex;
} catch (Exception $e) {
    http_response_code(500);
    echo "Error: " . $e->getMessage();
}
?>
